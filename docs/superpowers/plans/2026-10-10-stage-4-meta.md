```python
"""Agregados del meta (campeones, draft, lados, ritmo, regiones) para champions.json y meta.json.

Fuente: games de Oracle's Elixir (models.oracles.parse_rows). Dos muestras: las partidas de Worlds y,
mientras Worlds tenga pocas, las de las ligas en el parche anterior como guía.
"""
from collections import defaultdict

WORLDS_PATCH = "16.20"
GUIDE_PATCH = "16.18"     # parche anterior presente en los datos (no hay partidas en 16.19)
GUIDE_UNTIL_GAMES = 10    # con esta cantidad de partidas de Worlds, la guía deja de publicarse
FIRST_ROTATION = 6        # picks 1 a 6 (B1 R1 R2 B2 B3 R3): la primera rotación del draft
# Posición global de cada pick del equipo en el draft (pick1 del azul es el primero de todos)
PICK_SLOTS = {"blue": (1, 4, 5, 8, 9), "red": (2, 3, 6, 7, 10)}


def draft_slots(game: dict) -> tuple[list[tuple[int, str]], list[tuple[int, str]]]:
    """(picks con su posición global 1-10, bans con su fase 1-2). Los vacíos se ignoran."""
    picks, bans = [], []
    for side in ("blue", "red"):
        row = game["teams"][side]
        for i, slot in enumerate(PICK_SLOTS[side], start=1):
            if row.get(f"pick{i}"):
                picks.append((slot, row[f"pick{i}"]))
        for i in range(1, 6):
            if row.get(f"ban{i}"):
                bans.append((1 if i <= 3 else 2, row[f"ban{i}"]))  # bans 1-3: primera fase
    return picks, bans


def _rate(part: int, total: int) -> float | None:
    return round(part / total, 4) if total else None


def champion_stats(games: list[dict]) -> list[dict]:
    """Por campeón: picks, bans, presencia = (picks + bans) / partidas, winrate y prioridad de draft.

    Presencia mide cuánto le importa el campeón al meta: uno baneado siempre tiene 0 picks y 100% de presencia.
    Los picks salen del campeón de cada jugador (siempre está); el orden del draft, de pick1-5 (puede faltar).
    """
    count = defaultdict(lambda: {"picks": 0, "bans": 0, "wins": 0, "first": 0})
    for g in games:
        for side in ("blue", "red"):
            for row in g["players"][side].values():
                champ = row.get("champion")
                if champ:
                    count[champ]["picks"] += 1
                    count[champ]["wins"] += g["winner"] == side
        picks, bans = draft_slots(g)
        for _, champ in bans:
            count[champ]["bans"] += 1
        # Primera rotación: pickeado entre los 6 primeros o baneado en la primera fase (una vez por partida)
        early = {c for slot, c in picks if slot <= FIRST_ROTATION} | {c for phase, c in bans if phase == 1}
        for champ in early:
            count[champ]["first"] += 1
    n = len(games)
    rows = [{"champion": champ, "games": n, "picks": c["picks"], "bans": c["bans"],
             "presence": _rate(c["picks"] + c["bans"], n), "wins": c["wins"],
             "winrate": _rate(c["wins"], c["picks"]), "first_rotation": _rate(c["first"], n)}
            for champ, c in count.items()]
    return sorted(rows, key=lambda r: (-r["presence"], r["champion"]))
```

```python
from statistics import median
from typing import Callable

from models.elo import team_regions
from pipeline.stats import _is_worlds

PhaseFn = Callable[[str], str] | None


def phase_of(date: str, schedule: list[dict]) -> str:
    """Fase de una partida de Worlds según el calendario (Oracle's Elixir no la trae)."""
    starts = {}
    for m in schedule:
        if m["date"]:
            starts[m["phase"]] = min(starts.get(m["phase"], m["date"]), m["date"])
    day = date[:10]
    if "knockout" in starts and day >= starts["knockout"]:
        return "knockout"
    if "swiss" in starts and day >= starts["swiss"]:
        return "swiss"
    return "play-in"


def _split_by_phase(games: list[dict], phase_fn: PhaseFn, summarize) -> dict:
    by_phase: dict[str, list[dict]] = defaultdict(list)
    if phase_fn:
        for g in games:
            by_phase[phase_fn(g["date"])].append(g)
    return {"total": summarize(games), "by_phase": {p: summarize(gs) for p, gs in by_phase.items()}}


def _sides(games: list[dict]) -> dict:
    wins = sum(g["winner"] == "blue" for g in games)
    blue = _rate(wins, len(games))
    return {"games": len(games), "blue_wins": wins, "blue_winrate": blue,
            "red_winrate": None if blue is None else round(1 - blue, 4)}


def side_stats(games: list[dict], phase_fn: PhaseFn = None) -> dict:
    """Winrate de cada lado, total y por fase. Ojo: en Bo3/Bo5 el perdedor elige lado (confusión)."""
    return _split_by_phase(games, phase_fn, _sides)


def _num(value) -> float | None:
    return float(value) if value not in (None, "") else None


def _pace(games: list[dict]) -> dict:
    minutes, kills, kill_minutes = [], 0.0, 0.0
    for g in games:
        seconds = _num(g["teams"]["blue"].get("gamelength"))
        if seconds is None:
            continue  # sin duración no hay ritmo; se ignora, no cuenta como 0
        minutes.append(seconds / 60)
        k = [_num(g["teams"][s].get("kills")) for s in ("blue", "red")]
        if None not in k:
            kills += sum(k)
            kill_minutes += seconds / 60
    return {"games": len(games), "median_minutes": round(median(minutes), 1) if minutes else None,
            # Ratio de totales (kills totales / minutos totales), no promedio de ratios
            "kills_per_min": round(kills / kill_minutes, 4) if kill_minutes else None}


def pace_stats(games: list[dict], phase_fn: PhaseFn = None) -> dict:
    """Duración mediana (una partida eterna no la mueve, al promedio sí) y kills por minuto."""
    return _split_by_phase(games, phase_fn, _pace)


def region_stats(games: list[dict], regions: dict[str, str]) -> list[dict]:
    """Winrate de cada región solo contra otras regiones: en un LCK vs LCK la región gana siempre el 50%."""
    count = defaultdict(lambda: [0, 0])  # región → [partidas, victorias]
    for g in games:
        rb, rr = regions.get(g["blue"]), regions.get(g["red"])
        if rb is None or rr is None or rb == rr:
            continue
        for side, region in (("blue", rb), ("red", rr)):
            count[region][0] += 1
            count[region][1] += g["winner"] == side
    rows = [{"region": r, "games": n, "wins": w, "winrate": _rate(w, n)} for r, (n, w) in count.items()]
    return sorted(rows, key=lambda r: (-r["winrate"], r["region"]))


def build_meta(oe_games: list[dict], schedule: list[dict], worlds_start: str | None) -> dict:
    """Todo lo que publican champions.json y meta.json, por muestra ("worlds" y, si hace falta, "guide")."""
    regions = team_regions(oe_games)
    worlds = [g for g in oe_games if _is_worlds(g, worlds_start)]
    samples = {"worlds": worlds}
    if len(worlds) < GUIDE_UNTIL_GAMES:
        samples["guide"] = [g for g in oe_games if g["teams"]["blue"].get("patch") == GUIDE_PATCH]
    phase = lambda date: phase_of(date, schedule)
    return {
        "worlds_patch": WORLDS_PATCH,
        "guide_patch": GUIDE_PATCH if "guide" in samples else None,
        "worlds_games": len(worlds),
        "champions": {name: champion_stats(gs) for name, gs in samples.items()},
        "samples": {name: {
            "games": len(gs),
            # La guía son ligas: sus fases no son las de Worlds, así que solo el total
            "sides": side_stats(gs, phase if name == "worlds" else None),
            "pace": pace_stats(gs, phase if name == "worlds" else None),
            "regions": region_stats(gs, regions),
        } for name, gs in samples.items()},
    }
```

```python
def validate_meta(champions: dict[str, list[dict]]) -> list[str]:
    """Un campeón por muestra y porcentajes entre 0 y 1."""
    errors = []
    for sample, rows in champions.items():
        for name, n in Counter(r["champion"] for r in rows).items():
            if n > 1:
```

```ts
import type { Match } from './types';

export interface SwissRound {
  round: string;
  groups: { record: string; matches: Match[] }[];
}

export interface TeamRecord {
  wins: number;
  losses: number;
  status: 'qualified' | 'eliminated' | 'alive';
}

const TO_ADVANCE = 3; // 3 victorias clasifican, 3 derrotas eliminan
const PENDING = 'a confirmar';

// "Round 3" → 3, para recorrer las rondas en orden
function roundNumber(round: string): number {
  return Number(round.match(/\d+/)?.[0] ?? 0);
}

function swissMatches(matches: Match[]): Match[] {
  return matches.filter((m) => m.phase === 'swiss').sort((a, b) => roundNumber(a.round) - roundNumber(b.round));
}

function apply(table: Map<string, { wins: number; losses: number }>, m: Match): void {
  if (!m.winner || !m.team1 || !m.team2) return;
  const winner = m.winner === 'team1' ? m.team1.name : m.team2.name;
  const loser = m.winner === 'team1' ? m.team2.name : m.team1.name;
  const get = (name: string) => table.get(name) ?? { wins: 0, losses: 0 };
  table.set(winner, { ...get(winner), wins: get(winner).wins + 1 });
  table.set(loser, { ...get(loser), losses: get(loser).losses + 1 });
}

// Cada ronda agrupada por el récord que tenían los equipos ANTES de jugarla (en la suiza se cruzan iguales)
export function swissRounds(matches: Match[]): SwissRound[] {
  const table = new Map<string, { wins: number; losses: number }>();
  const byRound = new Map<string, Match[]>();
  for (const m of swissMatches(matches)) byRound.set(m.round, [...(byRound.get(m.round) ?? []), m]);

  const rounds: SwissRound[] = [];
  for (const [round, list] of byRound) {
    const groups = new Map<string, Match[]>();
    for (const m of list) {
      const rec = m.team1 ? table.get(m.team1.name) ?? { wins: 0, losses: 0 } : null;
      const key = m.team1 && m.team2 && rec ? `${rec.wins}-${rec.losses}` : PENDING;
      groups.set(key, [...(groups.get(key) ?? []), m]);
    }
    // Más victorias primero; los cruces sin definir, al final
    const order = (key: string) => (key === PENDING ? -1 : Number(key.split('-')[0]));
    rounds.push({ round, groups: [...groups].sort((a, b) => order(b[0]) - order(a[0])).map(([record, ms]) => ({ record, matches: ms })) });
    list.forEach((m) => apply(table, m)); // recién ahora cuentan los resultados de esta ronda
  }
  return rounds;
}

export function swissStandings(matches: Match[]): Map<string, TeamRecord> {
  const table = new Map<string, { wins: number; losses: number }>();
  for (const m of swissMatches(matches)) {
    for (const team of [m.team1, m.team2]) if (team && !table.has(team.name)) table.set(team.name, { wins: 0, losses: 0 });
    apply(table, m);
  }
  const result = new Map<string, TeamRecord>();
  for (const [name, r] of table) {
    const status = r.wins >= TO_ADVANCE ? 'qualified' : r.losses >= TO_ADVANCE ? 'eliminated' : 'alive';
    result.set(name, { ...r, status });
  }
  return result;
}
```

```ts
import type { Match } from './types';

const ORDER = ['Quarterfinals', 'Semifinals', 'Finals'];

// Rondas de eliminación en orden; una ronda que no conocemos (ej. desempate) va al final
export function bracketRounds(matches: Match[]): { round: string; matches: Match[] }[] {
  const byRound = new Map<string, Match[]>();
  for (const m of matches) {
    if (m.phase === 'knockout') byRound.set(m.round, [...(byRound.get(m.round) ?? []), m]);
  }
  const rank = (round: string) => (ORDER.includes(round) ? ORDER.indexOf(round) : ORDER.length);
  return [...byRound].sort((a, b) => rank(a[0]) - rank(b[0])).map(([round, ms]) => ({ round, matches: ms }));
}
```

```ts
import type { ChampionEntry, MetaSample, Phase, SampleId } from '../types';
import { champHex } from './player';

type Tab = 'campeones' | 'draft' | 'lados' | 'ritmo' | 'regiones';
const TABS: [Tab, string][] = [['campeones', 'Campeones'], ['draft', 'Draft'], ['lados', 'Lados'], ['ritmo', 'Ritmo'], ['regiones', 'Regiones']];
const PHASES: Phase[] = ['play-in', 'swiss', 'knockout'];

// Estado mientras la página esté abierta; la muestra se elige la primera vez según los datos
const state = { tab: 'campeones' as Tab, sample: null as SampleId | null, sort: 'presence' as keyof ChampionEntry, dir: 'desc' as 'asc' | 'desc' };

function seg(group: string, options: [string, string][], current: string, label: string): string {
  const buttons = options.map(([id, text]) => `<button type="button" data-${group}="${id}" aria-pressed="${id === current}">${escapeHtml(text)}</button>`).join('');
  return `<div class="seg" role="group" aria-label="${label}">${buttons}</div>`;
}

function empty(): string {
  return '<div class="state"><b>Todavía no hay partidas de Worlds 2026</b><span>El meta del torneo aparece después de las primeras partidas.</span></div>';
}

const COLUMNS: { id: keyof ChampionEntry; label: string; show: (c: ChampionEntry) => string }[] = [
  { id: 'presence', label: 'Presencia', show: (c) => pct(c.presence) },
  { id: 'picks', label: 'Picks', show: (c) => String(c.picks) },
  { id: 'bans', label: 'Bans', show: (c) => String(c.bans) },
  { id: 'winrate', label: '% Vict.', show: (c) => c.picks < FEW_GAMES && c.winrate !== null ? `${pct(c.winrate)} <span class="muted">· pocas</span>` : pct(c.winrate) },
];

function header(): string {
  return COLUMNS.map((c) => {
    const active = c.id === state.sort;
    const sortState = active ? (state.dir === 'desc' ? 'descending' : 'ascending') : 'none';
    return `<th class="num" aria-sort="${sortState}"><button type="button" class="sort" data-sort="${c.id}">${c.label}${active ? (state.dir === 'desc' ? ' ▼' : ' ▲') : ''}</button></th>`;
  }).join('');
}

function champions(rows: ChampionEntry[]): string {
  if (!rows.length) return empty();
  // El winrate de un campeón con menos de 5 picks no se ordena: con 3 de 3 = 100% engañaría
  const key = (c: ChampionEntry) => (state.sort === 'winrate' && c.picks < FEW_GAMES ? null : c[state.sort] as number | string | null);
  const body = sortBy(rows, key, state.dir).map((c) => `
    <tr><td class="champ-cell">${champHex(c.champion)}<b>${escapeHtml(c.champion)}</b></td>${COLUMNS.map((col) => `<td class="num">${col.show(c)}</td>`).join('')}</tr>`).join('');
  return `
    <p class="note">Presencia = (picks + bans) / partidas: cuánto le importa el campeón al meta. Uno que se banea siempre tiene 0 picks y 100% de presencia. El % de victorias con menos de ${FEW_GAMES} picks dice "pocas" y no se ordena: con tan pocas partidas es casi azar.</p>
    <div class="table-wrap"><table class="rank"><thead><tr><th>Campeón (${rows[0].games} partidas)</th>${header()}</tr></thead><tbody>${body}</tbody></table></div>`;
}

function draft(rows: ChampionEntry[]): string {
  if (!rows.length) return empty();
  const top = sortBy(rows, (c) => c.first_rotation, 'desc').slice(0, 20);
  const body = top.map((c, i) => `
    <tr><td class="num">${i + 1}</td><td class="champ-cell">${champHex(c.champion)}<b>${escapeHtml(c.champion)}</b></td>
      <td class="num">${pct(c.first_rotation)}</td><td class="num">${c.picks}</td><td class="num">${c.bans}</td></tr>`).join('');
  return `
    <p class="note">Prioridad = % de partidas en que el campeón se elige en la primera rotación (los 6 primeros picks) o se banea en la primera fase de bans. Mide qué se considera imprescindible, no qué gana más.</p>
    <div class="table-wrap"><table class="rank"><thead><tr><th class="num">#</th><th>Campeón</th><th class="num">Prioridad</th><th class="num">Picks</th><th class="num">Bans</th></tr></thead><tbody>${body}</tbody></table></div>`;
}

function phaseLabels(byPhase: Partial<Record<Phase, unknown>>): Phase[] {
  return PHASES.filter((p) => byPhase[p]);
}

function sides(s: MetaSample): string {
  const t = s.sides.total;
  if (!t.games) return empty();
  return `
    <div class="kpis">
      <div class="bezel"><div class="core kpi"><span class="label">Lado azul</span><b>${pct(t.blue_winrate)}</b><span class="muted">${t.blue_wins} de ${t.games}</span></div></div>
      <div class="bezel"><div class="core kpi"><span class="label">Lado rojo</span><b>${pct(t.red_winrate)}</b><span class="muted">${t.games - t.blue_wins} de ${t.games}</span></div></div>
    </div>
    ${phaseLabels(s.sides.by_phase).length ? '<div class="chart-box"><canvas id="meta-chart" aria-label="Victorias por lado en cada fase"></canvas></div>' : ''}
    <p class="note">Cuidado al leerlo: en Bo3 y Bo5 el que perdió la partida anterior elige lado, y casi siempre elige azul. Entonces el rojo queda más seguido en manos del equipo que viene ganando: el % mezcla la ventaja del lado con quién lo elige.</p>`;
}

function pace(s: MetaSample): string {
  const t = s.pace.total;
  if (!t.games) return empty();
  return `
    <div class="kpis">
      <div class="bezel"><div class="core kpi"><span class="label">Duración mediana</span><b>${minutes(t.median_minutes)}</b><span class="muted">${t.games} partidas</span></div></div>
      <div class="bezel"><div class="core kpi"><span class="label">Kills por minuto</span><b>${fmt(t.kills_per_min, 2)}</b><span class="muted">entre los dos equipos</span></div></div>
    </div>
    ${phaseLabels(s.pace.by_phase).length ? '<div class="chart-box"><canvas id="meta-chart" aria-label="Duración mediana por fase"></canvas></div>' : ''}
    <p class="note">Mediana y no promedio: es la partida "del medio". Una partida de 50 minutos tira el promedio para arriba; la mediana casi no se mueve.</p>`;
}

function regions(s: MetaSample): string {
  if (!s.regions.length) return '<div class="state"><b>Todavía no hay partidas entre regiones</b><span>Aparecen cuando se crucen equipos de ligas distintas.</span></div>';
  const body = s.regions.map((r) => `
    <tr class="${r.games < FEW_GAMES ? 'few' : ''}"><td class="team"><b>${escapeHtml(r.region)}</b>${r.games < FEW_GAMES ? '<span class="muted">pocas partidas</span>' : ''}</td>
      <td class="num">${r.games}</td><td class="num">${r.wins}-${r.games - r.wins}</td><td class="num">${pct(r.winrate)}</td></tr>`).join('');
  return `
    <p class="note">Solo cuentan partidas contra otras regiones: en un LCK contra LCK la región gana siempre exactamente el 50%, así que esas partidas no dicen nada.</p>
    <div class="table-wrap"><table class="rank"><thead><tr><th>Región</th><th class="num">PJ</th><th class="num">V-D</th><th class="num">% Vict.</th></tr></thead><tbody>${body}</tbody></table></div>`;
}

// Limpieza del render actual (gráfico + listener). Vive a nivel de módulo: un click re-renderiza sin pasar por
// main.ts, y main.ts tiene que poder limpiar el último render, no el primero (fuga de la Etapa 3).
let teardown: () => void = () => {};

export function renderMeta(el: HTMLElement, data: SiteData): () => void {
  teardown();
  const options = sampleOptions(data.meta);
  if (!state.sample || !options.some(([id]) => id === state.sample)) state.sample = defaultSample(data.meta);
  const sample = data.meta.samples?.[state.sample];
  const champs = data.champions?.samples?.[state.sample] ?? [];
  const body = !sample ? empty()
    : state.tab === 'campeones' ? champions(champs)
    : state.tab === 'draft' ? draft(champs)
    : state.tab === 'lados' ? sides(sample)
    : state.tab === 'ritmo' ? pace(sample)
    : regions(sample);
  const guide = state.sample === 'guide'
    ? `<p class="note">Guía: partidas de las ligas en el parche ${escapeHtml(data.meta.guide_patch ?? '')}, no de Worlds. Sirve de orientación hasta que el torneo tenga 10 partidas; después se va.</p>` : '';

  el.innerHTML = `
    <section class="stack">
      <span class="eyebrow">Parche ${escapeHtml(data.meta.worlds_patch ?? '—')} · datos de Oracle's Elixir</span>
      <h1>Meta</h1>
      ${seg('tab', TABS, state.tab, 'Secciones del meta')}
      ${options.length > 1 ? seg('sample', options, state.sample, 'Muestra') : ''}
      ${guide}
    </section>
    <section class="stack">${body}</section>`;

  // Gráfico de la sub-vista (si corresponde); se destruye al salir o re-renderizar
  let destroy: (() => void) | null = null;
  let alive = true;
  const canvas = el.querySelector<HTMLCanvasElement>('#meta-chart');
  if (canvas && sample) {
    const phases = phaseLabels(state.tab === 'lados' ? sample.sides.by_phase : sample.pace.by_phase);
    const labels = phases.map((p) => PHASE_LABELS[p]);
    const draw = state.tab === 'lados'
      ? drawBarChart(canvas, labels, [
          { label: 'Azul', data: phases.map((p) => sample.sides.by_phase[p]?.blue_winrate ?? null), color: '--blue' },
          { label: 'Rojo', data: phases.map((p) => sample.sides.by_phase[p]?.red_winrate ?? null), color: '--red' }],
        (v) => `${Math.round(v * 100)}%`)
      : drawBarChart(canvas, labels, [
          { label: 'Duración mediana', data: phases.map((p) => sample.pace.by_phase[p]?.median_minutes ?? null), color: '--cyan' }],
        (v) => minutes(v));
    draw.then((d) => { if (alive) destroy = d; else d(); });
  }

  el.onclick = (event) => {
    const button = (event.target as HTMLElement).closest<HTMLButtonElement>('button');
    if (!button) return;
    const d = button.dataset;
    if (d.tab) state.tab = d.tab as Tab;
    else if (d.sample) state.sample = d.sample as SampleId;
    else if (d.sort) {
      state.dir = state.sort === d.sort && state.dir === 'desc' ? 'asc' : 'desc';
      state.sort = d.sort as keyof ChampionEntry;
    } else return;
    renderMeta(el, data);
  };
  teardown = () => {
    alive = false;
    destroy?.();
    el.onclick = null;
  };
  return () => teardown();
}
```
